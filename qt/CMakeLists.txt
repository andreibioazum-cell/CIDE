cmake_minimum_required(VERSION 3.21)
project(CIDEQt VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick WebView)
qt_standard_project_setup()
qt_add_executable(CIDEQt src/main.cpp src/editorcontroller.cpp src/editorcontroller.h)
qt_add_qml_module(CIDEQt URI CIDE VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(CIDEQt PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::WebView)
set_target_properties(CIDEQt PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
